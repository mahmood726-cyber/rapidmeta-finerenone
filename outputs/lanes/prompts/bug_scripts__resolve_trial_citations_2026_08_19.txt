#!/usr/bin/env python3
"""RESOLVE THE PRIMARY PUBLICATION FOR EACH NAMED TRIAL -- BY LOOKUP, NEVER FROM RECALL.

WHY THE RULE IS ABSOLUTE HERE. This produces a reference list for submission. Somebody will
check it. `identifiers from recall` is the class this project has hit repeatedly -- most
recently a PMID that named an atrial-fibrillation trial in a colchicine benchmark, and the
corpus held the evidence to refute it the whole time (DEFECT-REGISTRY class 22).

    EVERY FIELD BELOW COMES FROM A FETCH. Acronyms are read from the registration's own
    `identificationModule`; candidate publications from its `referencesModule`; and author,
    journal, volume, pages and DOI from PubMed. Nothing is typed from memory, and a trial whose
    primary publication cannot be resolved is reported UNRESOLVED rather than approximated.

WHAT COUNTS AS THE PRIMARY PUBLICATION, and the limit of the method
    The registry types each reference `RESULT` (the sponsor's own results citation), `DERIVED`
    (auto-linked from PubMed) or `BACKGROUND`. This prefers RESULT, then DERIVED, and reports
    HOW MANY candidates it saw. IT DOES NOT READ THE PAPERS. A registration listing several
    result publications -- a main paper and its substudies -- yields several candidates, and
    choosing among them by position would be P35 one level down. Where there is more than one,
    all are listed and the choice is left to a human.

USAGE:  python scripts/resolve_trial_citations_2026_08_19.py
"""

def resolve(nct):
    st, s, detail = X.fetch_raw(nct, fields="protocolSection,derivedSection")
    if st != X.OK:
        return {"state": "UNREACHABLE", "detail": str(detail)[:160]}
    ps = s.get("protocolSection") or {}
    idm = ps.get("identificationModule") or {}
    refs = (ps.get("referencesModule") or {}).get("references") or []
    cands = []
    for r in refs:
        t = str(r.get("type") or "").upper()
        if t in ("RESULT", "DERIVED"):
            cands.append({"pmid": r.get("pmid"), "type": t,
                          "citation_as_the_registry_prints_it": (r.get("citation") or "")[:300]})
    return {
        "state": "OK",
        "acronym_as_the_registration_declares_it": idm.get("acronym"),
        "brief_title": (idm.get("briefTitle") or "")[:150],
        "official_title": (idm.get("officialTitle") or "")[:220],
        "org_study_id": (idm.get("orgStudyIdInfo") or {}).get("id"),
        "overall_status": (ps.get("statusModule") or {}).get("overallStatus"),
        "enrolment": ((ps.get("designModule") or {}).get("enrollmentInfo") or {}).get("count"),
        "n_result_or_derived_references": len(cands),
        "candidate_publications": cands,
    }


def main():
    out = {"resolved_utc": "2026-08-19",
           "rule": ("Every field from a lookup. Acronym read from the registration; candidate "
                    "publications from its referencesModule; bibliographic detail from PubMed. "
                    "An unresolvable citation is reported UNRESOLVED, never approximated."),
           "trials": {}}
    for name, nct, prov in TRIALS:
        if not nct:
            out["trials"][name] = {"registration": None, "provenance": prov,
                                   "state": "NO REGISTRATION RESOLVED YET"}
            print("%-26s %-14s %s" % (name, "-", prov))
            continue
        r = resolve(nct)
        r["registration"] = nct
        r["registration_provenance"] = prov
        out["trials"][name] = r
        ac = r.get("acronym_as_the_registration_declares_it")
        flag = ""
        if r["state"] == "OK":
            base = name.split(" (")[0].split(" / ")[0].upper().replace("-", "")
            if ac and base not in str(ac).upper().replace("-", ""):
                flag = "  <-- ACRONYM MISMATCH, check"
            elif not ac:
                flag = "  <-- registration declares NO acronym"
        print("%-26s %-14s acronym=%-18s refs=%s%s"
              % (name, nct, str(ac), r.get("n_result_or_derived_references"), flag))
    with io.open(DEST, "w", encoding="utf-8") as fh:
        fh.write(json.dumps(out, indent=1, ensure_ascii=False))
    print("\nwrote %s" % os.path.relpath(DEST, REPO))
    return 0


if __name__ == "__main__":
    sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8", errors="replace")
    sys.exit(main())
