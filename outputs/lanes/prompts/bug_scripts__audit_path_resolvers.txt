THAT IS WORSE THAN A WRONG ADDRESS. A path pointing somewhere that does not exist fails

# A resolver: a def whose body splits a path and iterates the segments.
DEF = re.compile(r"^([ 	]*)def\s+(\w+)\s*\(([^)]*)\)\s*:", re.M)
# `\s*` NOT `[ 	]*` -- the first version used \s*, which under re.M consumed the
# preceding NEWLINES into the indent capture, so every extracted body came back as two
# empty lines and the sweep reported ZERO resolvers across 782 files. It returned
# NOT_ASSESSABLE rather than a clean bill, which is the only reason it was noticed --
# a matcher that matched and yielded nothing, in the file written to sweep for matchers
# that resolve to the wrong thing.
SPLITS = re.compile(r"\.split\(\s*['\"]\.['\"]\s*\)|\.split\(\s*['\"]/['\"]\s*\)")
WALKS = re.compile(r"\bfor\s+\w+(?:\s*,\s*\w+)?\s+in\s+")


def body_of(src, start, indent):
    """The lines of a def, by indentation."""
    lines = src[start:].split("\n")
    out = [lines[0]]
    for ln in lines[1:]:
        if ln.strip() and not ln.startswith(indent + " ") and not ln.startswith(indent + "\t"):
            break
        out.append(ln)
    return "\n".join(out)


def main():
    resolvers, flagged = 0, []
    files = 0
    for root in SEARCH:
        if not os.path.isdir(root):
            continue
        for dirpath, _dirs, names in os.walk(root):
            for nm in sorted(names):

                    continue

                try:
                    src = io.open(fp, encoding="utf-8", errors="replace").read()

                    continue
                files += 1

                    indent, name = m.group(1), m.group(2)
                    body = body_of(src, m.start(), indent)
                    if not SPLITS.search(body) or not WALKS.search(body):
                        continue
                    resolvers += 1
                    reasons = []
                    # Does it exit the walk without consuming every segment?
                    walk_at = WALKS.search(body).start()
                    tail = body[walk_at:]
                    if re.search(r"^\s+return\b(?!\s*(None|obj|cur)\s*$)", tail, re.M):
                        reasons.append("returns from inside the walk")
                    if re.search(r"^\s+break\b", tail, re.M):
                        reasons.append("breaks out of the walk")
                    if re.search(r"['\"]\*['\"]", body):
                        reasons.append("has a wildcard branch")
                    if reasons:
                        rel = os.path.relpath(fp, REPO).replace("\\", "/")
                        flagged.append((rel, name, "; ".join(reasons)))

    print("python files read              %d" % files)
    print("path resolvers found           %d" % resolvers)
    print("with the container-return shape %d" % len(flagged))
    print()

            print("    shape flagged: %s" % why)
