# `pooled_OR: 0.500065` is rendered on dashboard.html under "Pooled OR (95% CI)". NEGATIVE is a
# sidecar with no placeholder comment upstream, which must not be flagged.

WHEN YOU FIND A COMMENT WARNING ABOUT A VALUE, ASSUME IT IS PUBLISHED UNTIL YOU PROVE

    the sidecar     pooled_OR 0.500065, ci 0.396901-0.630043, tau2 0.0, I2 0.0
    the index       outputs/portfolio_index.json carries the row
    the reader      dashboard.html renders it under "Pooled OR (95% CI)"

`tau2 = 0.0` and `I2 = 0.0` are the tell: three numbers chosen to look plausible agree

AND THE FLAG THE COMMENT PROMISED EXISTS AND LIES. "These will be flagged for human

THIS DOES NOT REPAIR ANYTHING INTO A REAL NUMBER. It finds every value whose source disowns it

from __future__ import annotations

import glob
import io
import json
import os
import re
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from instrument_controls import require_controls          # noqa: E402

REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
OUT = os.path.join(REPO, "outputs", "self_disowning_values_2026_08_23.json")

DISOWN = re.compile(
    r"(?i)(placeholder|synthetic|illustrative|dummy|made[- ]up|invented|"
    r"for demonstration|example only|not real|fabricat|approximat(?:e|ion) only|"
    r"will be flagged|needs? human verification|to be replaced)")

# Where a generated value can end up in front of a reader.
SURFACES = ["dashboard.html", "index.html", "audit_table.html", "portfolio_pools.html"]


def sources():
    for pat in ("scripts/*.R", "scripts/*.py", "*.py"):
        for p in sorted(glob.glob(os.path.join(REPO, pat))):
            yield p


def disowning_lines(path):
    out = []
    try:
        src = io.open(path, encoding="utf-8", errors="replace").read()
    except OSError:
        return out
    for i, line in enumerate(src.splitlines(), 1):
        if DISOWN.search(line) and (line.strip().startswith("#") or "#" in line):
            out.append((i, re.sub(r"\s+", " ", line.strip())[:120]))
    return out


def sidecars_named_near(path, lineno, window=40):
    """Output files named within a few lines of a disowning comment."""
    try:
        lines = io.open(path, encoding="utf-8", errors="replace").read().splitlines()
    except OSError:
        return []
    seg = "\n".join(lines[max(0, lineno - window):lineno + window])
    return sorted(set(re.findall(r'["\']([\w/\.\-]+\.json)["\']', seg)))


def main():
    sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8", errors="replace")

    r = os.path.join(REPO, "scripts", "regenerate_catastrophic_sidecars.R")
    fgfr_flagged = any("placeholder" in t.lower() for _l, t in disowning_lines(r))
    clean_src = os.path.join(REPO, "scripts", "instrument_controls.py")
    require_controls(
        "self_disowning_values",
        ("regenerate_catastrophic_sidecars.R disowns its own HRs as placeholders",
         fgfr_flagged, True),
        ("a file with no placeholder comment is not flagged",
         bool(disowning_lines(clean_src)), True))

    findings = []
    for p in sources():
        rel = os.path.relpath(p, REPO).replace("\\", "/")
        for line, text in disowning_lines(p):
            findings.append({"file": rel, "line": line, "comment": text,
                             "writes": sidecars_named_near(p, line)})

    # which of the named outputs actually exist, and which surfaces carry their numbers
    reached = []
    idx_path = os.path.join(REPO, "outputs", "portfolio_index.json")
    idx = json.load(io.open(idx_path, encoding="utf-8")) if os.path.isfile(idx_path) else []
    rows = idx if isinstance(idx, list) else idx.get("rows", [])
    for f in findings:
        for w in f["writes"]:
            full = os.path.join(REPO, w)
            if not os.path.isfile(full):
                continue
            try:
                side = json.load(io.open(full, encoding="utf-8"))
            except ValueError:
                continue
            if not isinstance(side, dict):
                continue          # a list-shaped output holds no single pooled estimate
            val = side.get("pooled_OR")
            if val is None:
                continue
            hit = [rr.get("topic") for rr in rows if isinstance(rr, dict)
                   and isinstance(rr.get("pooled_OR"), (int, float))
                   and abs(rr["pooled_OR"] - val) < 1e-6]
            reached.append({"sidecar": w, "pooled_OR": val,
                            "regenerated_from": side.get("regenerated_from"),
                            "tau2": side.get("tau2"), "I2": side.get("I2"),
                            "in_portfolio_index": hit,
                            "from": "%s:%d" % (f["file"], f["line"])})

    print("")
    print("SELF-DISOWNING VALUES: a source that calls its own number a placeholder")
    print("")
    print("   disowning comments found            %3d" % len(findings))
    print("   whose output EXISTS and holds a     %3d" % len(reached))
    print("      pooled estimate")
    print("")
    for f in findings[:14]:
        print("   %-46s:%-5d %s" % (f["file"][:46], f["line"], f["comment"][:58]))
        if f["writes"]:
            print("        writes: %s" % ", ".join(f["writes"][:3]))
    print("")
    print("   REACHING A READER:")
    for rr in reached:
        print("      %-44s pooled_OR %-10s tau2 %-5s I2 %-5s"
              % (rr["sidecar"][:44], rr["pooled_OR"], rr["tau2"], rr["I2"]))
        print("         regenerated_from: %r" % rr["regenerated_from"])
        print("         in portfolio_index -> dashboard: %s" % (rr["in_portfolio_index"] or "no"))
    print("")
    print("tau2 = 0 AND I2 = 0 TOGETHER ARE THE TELL: numbers chosen by hand agree perfectly,")
    print("because nothing generated them. A real three-trial pool almost never does.")
    if not os.path.isdir(os.path.dirname(OUT)):
        os.makedirs(os.path.dirname(OUT))
    json.dump({"disowning": findings, "reaching_a_reader": reached},
              io.open(OUT, "w", encoding="utf-8"), indent=1)
    if reached:
        sys.exit("REFUSED: %d value(s) whose own source calls them placeholders are rendered "
                 "as findings." % len(reached))


if __name__ == "__main__":
    main()
