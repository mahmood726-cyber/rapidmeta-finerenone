def norm(s):
    return " ".join(str(s or "").lower().replace("–", "-").split())


def _any(terms, blob):
    return sorted({t for t in terms if t in blob})


def assign(study):
    ps = study.get("protocolSection") or {}
    idm = ps.get("identificationModule") or {}
    dm = ps.get("designModule") or {}
    conds = norm(" ; ".join((ps.get("conditionsModule") or {}).get("conditions") or []))
    title = norm(" ".join([str(idm.get("briefTitle") or ""), str(idm.get("officialTitle") or "")]))
    both = conds + " " + title
    role, role_ev = TI.locate(study, TI.synonyms_for("dabigatran")
                              if "dabigatran" in TI.TOPIC_SYNONYMS
                              else ["dabigatran", "dabigatran etexilate", "pradaxa", "bibr 1048"])
    ev = {
        "brief_title": (idm.get("briefTitle") or "")[:130],
        "conditions": (ps.get("conditionsModule") or {}).get("conditions"),
        "enrolment": (dm.get("enrollmentInfo") or {}).get("count"),
        "status": (ps.get("statusModule") or {}).get("overallStatus"),
        "allocation": (dm.get("designInfo") or {}).get("allocation"),
        "n_arms": len((ps.get("armsInterventionsModule") or {}).get("armGroups") or []),
        "has_results": bool(study.get("hasResults")),
        "arm_role": role,
        "phases": dm.get("phases"),
    }
    hit = _any(CEREBRAL, both)
    if hit:
        return "CEREBRAL", "names %s" % ", ".join(hit), ev
    hit = _any(SURGICAL, both)
    if hit:
        return "SURGICAL", "the population is an elective arthroplasty (%s)" % ", ".join(hit), ev
    hit = _any(EXTENDED, both)
    if hit:
        return "EXTENDED", "randomises after a completed course (%s)" % ", ".join(hit), ev
    hit = _any(TREATMENT, both)
    if hit:
        return "TREATMENT", "acute symptomatic venous thromboembolism (%s)" % ", ".join(hit), ev
    return "NONE", "no reading's population terms are matched by the conditions or the title", ev


def disposition(reading, ev):
    """EVERY failing limb is reported, never only the first (P15)."""
    fails = []
    if reading == "NONE":
        fails.append("POPULATION: outside all four readings")
    if ev["arm_role"] == "not_assessable":
        return "NOT_ASSESSABLE", ["ARM ROLE: dabigatran could not be located in the arms, the "
                                  "interventions or the title. ABSENT IS NOT ZERO."]
    if ev["arm_role"] not in ("experimental", "comparator"):
        fails.append("INTERVENTION: dabigatran is not the randomised contrast (role %r)"
                     % ev["arm_role"])
    if (ev["allocation"] or "").upper() != "RANDOMIZED":
        fails.append("DESIGN: allocation is %r, not RANDOMIZED" % ev["allocation"])
    if ev["n_arms"] < 2:
        fails.append("COMPARATOR: %d arm(s) declared; no randomised contrast to read"
                     % ev["n_arms"])
    if fails:
        return "EXCLUDED", fails
    st = (ev.get("status") or "").upper()
    if not ev["has_results"]:
        if st in ("COMPLETED", "TERMINATED", "WITHDRAWN", "SUSPENDED", "UNKNOWN"):
            return "ELIGIBLE_COMPLETED_NO_RESULTS_POSTED", [
                "eligible on every limb; status %r and NO results section on this registry. A "
                "publication may exist and this registry search cannot see one." % ev["status"]]
        return "ELIGIBLE_NO_RESULTS_YET", ["eligible; status %r, not yet reported" % ev["status"]]
    return "ELIGIBLE_WITH_RESULTS", ["eligible on every limb, and results are posted. The "
                                     "estimand screen is a SEPARATE stage."]


def screen():
    rows = []
    for nct in SURFACED:
        st, s, detail = X.fetch_raw(nct, fields="protocolSection,hasResults")
        if st != X.OK:
            rows.append({"nct": nct, "reading": "NOT_ASSESSABLE",
                         "why": "transport: %s" % st, "disposition": "NOT_ASSESSABLE",
                         "disposition_why": ["TRANSPORT %s -- unreachable is not absent" % st],
                         "evidence": {}})
            continue
        reading, why, ev = assign(s)
        disp, dwhy = disposition(reading, ev)
        rows.append({"nct": nct, "reading": reading, "why": why, "disposition": disp,
                     "disposition_why": dwhy, "evidence": ev})
    return rows


def assert_known(rows):

    bad = []
    for nct, (want, why) in sorted(KNOWN_MEMBERS.items()):
        r = by.get(nct)
        if r is None:
            bad.append("%s is a declared known member and IS NOT IN THE SURFACED SET" % nct)
        elif r["reading"] != want:
            bad.append("%s -> %s, declared %s (%s)" % (nct, r["reading"], want, why))
    return bad

def selftest():
    fails = []

    def ck(name, got, want):
        ok = got == want
        print("  %-66s %s  %r" % (name, "ok" if ok else "FAIL", got))

        if st != X.OK:

            continue
        ck("%s -> %s" % (nct, want), assign(s)[0], want)

    print("\n2. THE GUARD CAN FIRE, and does not fire on the correct case (P16):")
    # ---------------------------------------------------------------------------------------
    # READ THIS BEFORE ASSUMING THE GUARD WAS LOOSENED. It was not. `assert_known` was never
    # changed; the ASSERTION was.
    #
    # The first version of this block passed a ONE-ROW list to `assert_known`, which checks all
    # seven declared members. Six were legitimately reported ABSENT, so the call returned 7
    # while the assertion expected 1, and the selftest failed.
    #
    #   THE TEST WAS WRONG AND THE GUARD WAS RIGHT.
    #
    # That is the harder of the two cases a control exists to tell apart, and the tempting fix

    # only thing protecting the partition: its ability to notice a member that is missing rather

    # A future reader who finds a guard and a test disagreeing should establish which one is

                 for r in allright]
    ck("exactly one misplaced member is reported", len(assert_known(one_wrong)), 1)
