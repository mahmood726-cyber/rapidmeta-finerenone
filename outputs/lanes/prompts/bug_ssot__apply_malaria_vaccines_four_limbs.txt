def main():
    dry = "--apply" not in sys.argv
    path = os.path.join(REPO, "ssot", TOPIC, TOPIC + ".json")
    obj = json.load(io.open(path, encoding="utf-8"))
    by_outcome = (obj.get("results") or {}).get("by_outcome") or {}
    declared = [d.get("id") for d in (obj.get("outcomes") or []) if isinstance(d, dict)]

    # ---- limb 1: risk of bias, PER RESULT (every trial in every block it appears in) -----
    rob_by_outcome = {}
    n_results = 0
    for oid, blk in sorted(by_outcome.items()):
        if not isinstance(blk, dict):
            continue
        if oid not in declared:
            sys.exit("REFUSED: `%s` is not declared in outcomes[], so anything attached to "
                     "it would not render." % oid)
        per = {}
        for t in blk.get("per_trial") or []:
            tid = t.get("trial_id")
            spec = TRIALS.get(tid)
            if not spec:
                sys.exit("REFUSED: no registration was read for `%s` on outcome `%s`. An "
                         "unassessed result inside an assessed outcome is exactly the gap "
                         "'per result' exists to close." % (tid, oid))
            if t.get("nct") and t["nct"] != spec["nct"]:
                sys.exit("REFUSED: %s carries nct %r on `%s` and the registration read here "
                         "was %r." % (tid, t.get("nct"), oid, spec["nct"]))
            entry = {
                "nct": spec["nct"], "trial": spec["name"], "cohort_id": tid,
                "registered_enrolment": spec["n"],
                "registered_masking": spec["masking"],
                "registered_sites": spec["sites"],
                "result_assessed": "the result this object carries for outcome `%s`" % oid,
                "domains": {
                    "D1_randomisation_process": D1,
                    "D2_deviations_from_intended_intervention": D2,
                    "D3_missing_outcome_data": D3,
                    "D4_measurement_of_the_outcome": spec["d4"],
                    "D5_selection_of_the_reported_result": spec["d5"],
                },
                "overall": "SOME_CONCERNS",
                "overall_reason": (
                    "D2 and D3 are NO_INFORMATION on every cohort here, and under RoB 2 an "
                    "unjudgeable domain cannot yield LOW overall. Where D4 or D5 is also "
                    "SOME_CONCERNS the reason is named in that domain."),
            }
            if spec.get("extra"):
                entry["also_recorded"] = spec["extra"]
            per[spec["nct"] + "::" + tid] = entry
            n_results += 1
        if per:
            rob_by_outcome[oid] = per

    atomic_write.merge_not_overwrite(obj, "risk_of_bias", {
        "tool": "RoB 2 (Cochrane risk-of-bias tool for randomized trials)",
        "assessed_utc": TODAY,
        "assessed_per": "RESULT, not trial -- Handbook 8.2. A cohort appearing in three "
                        "outcome blocks is assessed three times, because the domains that "
                        "matter differ by the result being used.",
        "by_outcome": rob_by_outcome,
        "sources_read": ["ClinicalTrials.gov API v2 %s -- design module, eligibility module, "
                         "location module and registered primary outcomes" % n
                         for n in sorted({s["nct"] for s in TRIALS.values()})],
        "sources_NOT_read": ("The trial publications. D2 and D3 depend on them and are "
                             "NO_INFORMATION for that reason."),
        "ceiling": CEILING,
        "TWO_COHORTS_ARE_OPEN_LABEL": (
            "NCT03276962 and NCT00436007 both register MASKING: NONE, on an endpoint that is "
            "clinical malaria against a case definition. NCT03276962 CONTRIBUTES TO A POOLED "
            "ESTIMATE, so an open-label result reaches a pooled number on this topic."),
        "ONE_COHORT_REGISTERS_NO_EFFICACY_PRIMARY": (
            "NCT00436007's ONLY registered primary outcome is SERIOUS ADVERSE EVENTS. The "
            "result carried from it here is an efficacy outcome, which is not a registered "
            "primary on that trial at all. Same shape as NCT00081744 on tigecycline-ciai "
            "(open item O0a)."),
        "ONE_ASSESSOR_ONLY": (
            "ASSESSED BY ONE ASSESSOR. Under the standing specification that risk of bias 2 "
            "must be complete and done by TWO AIs from different model families, THIS "
            "ASSESSMENT IS INCOMPLETE. Recorded as such, and `rob2.assessors` is deliberately "
            "NOT written."),
    }, STAMP)

    # ---- limb 2: GRADE, per POOLED outcome ----------------------------------------------
    pooled_ids = sorted(oid for oid, blk in by_outcome.items()
                        if isinstance(blk, dict)
                        and isinstance(blk.get("pooled"), dict)
                        and blk["pooled"].get("point") is not None)
    if sorted(GRADE) != pooled_ids:
        sys.exit("REFUSED: GRADE is written for %r and the object's POOLED outcomes are %r. "
                 "Rating an outcome that is not pooled, or leaving a pooled one unrated, is "
                 "the gap this limb exists to close." % (sorted(GRADE), pooled_ids))
    atomic_write.merge_not_overwrite(obj, "grade", {
        "approach": ("GRADE, following the Cochrane Handbook chapter 14. Randomised evidence "
                     "starts HIGH and is rated down with reasons."),
        "assessed_utc": TODAY,
        "by_outcome": dict(
            (oid, {"k": g["k"], "starting_certainty": "HIGH", "certainty": g["certainty"],
                   "steps": g["steps"], "summary": g["summary"],
                   "rating_up_not_applied": (
                       "No domain is rated UP. Rating up for a large effect, a dose-response "
                       "gradient or opposing plausible confounding applies to observational "
                       "evidence; these are randomised trials, which start HIGH.")})
            for oid, g in GRADE.items()),
        "outcomes_NOT_rated_and_why": (
            "SEVEN of this object's nine outcome blocks carry NO POOLED ESTIMATE -- they are "
            "single-cohort results. GRADE here is applied PER POOLED OUTCOME, per the page "
            "standard, so those seven are not rated. Their risk of bias IS assessed, per "
            "result, above."),
    }, STAMP)

    # ---- limb 3: published comparison ----------------------------------------------------
    d = dict(SCREEN["denom"])
    d["_house_form"] = (
        "matched / retrieved / read / appraised / not returned -- P53. %d flagged by title; "
        "%d appraised against their abstracts and every other record NOT READ beyond its "
        "title and summary." % (d["flagged_by_title"], d["appraised"]))
    atomic_write.merge_not_overwrite(obj, "published_comparison", {
        "_why": "P46 limb 3.",
        "_how_identified": (
            "PubMed E-utilities esearch + esummary, executed %s. Query, counts and per-record "
            "disposition in ssot/%s/appraisal/PUBLISHED_SYNTHESIS_SCREEN.json." % (TODAY, TOPIC)),
        "denominator": d,
        "identity_basis": ("NEITHER APPRAISED REVIEW NAMES ITS INCLUDED STUDIES and no "
                           "included-study table was opened, so both counts are COUNTED AND "
                           "NOT IDENTIFIED."),
        "reviews": SCREEN["reviews"],
        "THE_FINDING_OF_THIS_COMPARISON_%s" % STAMP:
            " ".join(SCREEN["finding"][k] for k in sorted(SCREEN["finding"])),
    }, STAMP)

    # ---- limb 4: model output, verbatim, at the HOUSE KEY --------------------------------
    for oid in pooled_ids:
        fit = os.path.join(FITS, "%s.txt" % oid)
        if not os.path.exists(fit):
            sys.exit("REFUSED: no stored R output at %s." % fit)
        text = io.open(fit, encoding="utf-8").read()
        if "AGREES WITH THE STORED POINT TO 4 dp: YES" not in text:
            sys.exit("REFUSED: the stored fit for %s does not record agreement with the "
                     "stored point." % oid)
        if "HOUSE INTERVAL, inflation factor floored at 1" not in text:
            sys.exit("REFUSED: the stored fit for %s predates the Hartung-Knapp floor fix and "
                     "would quote an interval this project's own rule forbids. Re-run "
                     "ssot/fit_from_per_trial.R." % oid)
        by_outcome[oid]["r_output"] = {
            "state": "PRESENT",
            "what_it_is": ("The random-effects fit of THIS BLOCK'S OWN per-trial estimates, "
                           "quoted as the software printed it."),
            "script": "ssot/fit_from_per_trial.R",
            "call": "Rscript ssot/fit_from_per_trial.R %s %s" % (TOPIC, oid),
            "environment": "R 4.6.0 with metafor 5.0.1",
            "run_utc": TODAY,
            "interval_method": (
                "REML with the unadjusted Wald interval, and the same fit under Hartung-Knapp "
                "beside it. THE HOUSE INTERVAL FLOORS THE VARIANCE-INFLATION FACTOR AT 1 so "
                "the adjustment can never narrow below the unadjusted interval; metafor's raw "
                "unfloored knha value is printed separately and flagged where it is narrower."),
            "reproduction_of_the_previous_value": (
                "REPRODUCES THE POINT ESTIMATE THIS PAGE DELIVERS TO FOUR DECIMAL PLACES."),
            "verbatim": text,
            "stored_at": "evidence/malaria_fits_2026_08_21/%s.txt" % oid,
        }

        prior = by_outcome[oid].get("POOL_FINDINGS_%s" % STAMP) or {}
        prior.update(FINDINGS_ON_POOL[oid])
        by_outcome[oid]["POOL_FINDINGS_%s" % STAMP] = prior

    obj.setdefault("display_change_announced", []).append({
        "date": TODAY,
        "change": "all four P46 limbs written; this topic was at 0 of 4",
        "values_moved": "NONE",
        "what_changed": ("risk of bias on %d results across %d outcome(s); GRADE on %d pooled "
                         "outcome(s); a published comparison over %d matched records; the R "
                         "output stored verbatim for both pools, each reproducing its stored "
                         "point to 4 dp"
                         % (n_results, len(rob_by_outcome), len(pooled_ids), d["matched"])),
        "why": "Every limb was ABSENT.",
    })

    print("%-20s rob %d results over %d outcome(s)" % (TOPIC, n_results, len(rob_by_outcome)))
    print("%-20s GRADE %s" % ("", ", ".join("%s=%s" % (o, GRADE[o]["certainty"])
                                            for o in pooled_ids)))
    print("%-20s comparison %d matched / %d read / %d appraised"
          % ("", d["matched"], d["read"], d["appraised"]))
    print("%-20s model output stored for %d pooled outcome(s)" % ("", len(pooled_ids)))

    if dry:
        print("DRY RUN -- pass --apply to write")
        return
    screen = os.path.join(REPO, "ssot", TOPIC, "appraisal", "PUBLISHED_SYNTHESIS_SCREEN.json")
    os.makedirs(os.path.dirname(screen), exist_ok=True)
    atomic_write.write_json(screen, {
        "executed_utc": TODAY,
        "source": "PubMed E-utilities esearch + esummary",
        "query_as_executed": SCREEN["query"],
        "matched": d["matched"], "retrieved": d["retrieved"], "read": d["read"],
        "flagged_by_title": d["flagged_by_title"],
        "appraised": [r["pmid"] for r in SCREEN["reviews"]],
        "not_returned_by_the_tool": 0,
        "_honesty": ("Two records appraised against their abstracts; every other "
                     "title-flagged record and every unflagged summary NOT READ beyond its "
                     "title. No included-study table was opened and no full text retrieved."),
    }, indent=1)
    atomic_write.write_json(path, obj, indent=1)


if __name__ == "__main__":
    main()
