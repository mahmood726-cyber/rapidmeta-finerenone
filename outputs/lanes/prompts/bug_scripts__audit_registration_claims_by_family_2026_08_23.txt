"""Registration claims, split by family -- because only one of the two families is defective.

# control: routed through require_controls with pages a census lane read. POSITIVE is
# BARIATRIC_RYGB_VS_SG, reported as carrying a BARE UNVERIFIED COUNT with no identifier (47).
# NEGATIVE is that the *_AUTO_FULL family must NOT be counted as making an unsupportable claim,
# because it answers "No" honestly -- the error this script exists to avoid.

REPORTING WHAT THE CORPUS DOES RIGHT IS PART OF THE MEASUREMENT, NOT A COURTESY. An earlier
count said "422 pages assert a registration claim" and treated the whole set as one defect.
That was wrong twice over, and the split is the finding:

  THE HONEST FAMILY -- `*_AUTO_FULL`. Answers AMSTAR 2 item 2 with **No**: "Retrospective
  public protocol pack ... not prospectively registered." That is a correct answer to a
  question about itself, volunteered where a false Yes would never have been checked. IT NEEDS
  NOTHING DONE TO IT, and a repair programme that rewrites it would be removing the one thing
  on the page that was already right.

  THE DEFECTIVE FAMILY -- hand-built NMA / broad reviews. Answers **Yes**, self-certified via
  "This protocol tab (living protocol)", with no external registration identifier. A protocol
  that registers itself, after the fact, on the page whose method it is describing, is not a
  registration -- it is the claim restating itself. Only this group is making an unsupportable
  claim and only this group is counted as a defect.

AND A WORSE VARIANT OF THE UNVERIFIED-IDENTIFIER PROBLEM, COUNTED SEPARATELY BECAUSE THE
REASONING SHOULD BE TESTED RATHER THAN TAKEN:

    "156 number(s) on this page marked UNVERIFIED -- no resolvable trial id"

The argument for separating these is that a NAMED unresolvable NCT can be looked up by a
reader and argued with, whereas a bare COUNT cannot be checked at all -- it reports a quantity
of doubt without exposing anything to inspect. This run prints both populations and the
overlap so the distinction can be inspected rather than assumed; if a page turns out to carry
BOTH a bare count and named ids, the argument is weaker than it looks and the overlap column
is where that shows.
"""
from __future__ import annotations

import collections
import io
import json
import os
import re
import subprocess
import sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from instrument_controls import require_controls  # noqa: E402

REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
REF = "origin/main"
CLASSES = os.path.join(REPO, "outputs", "five_bucket_classes_2026_08_23.json")
OUT = os.path.join(REPO, "outputs", "registration_claims_by_family_2026_08_23.json")

# AMSTAR 2 item 2, answered Yes, self-certified by pointing at the page's own protocol tab.
SELF_CERT = re.compile(
    r"(?is)protocol registered before commencement.{0,400}?\byes\b.{0,300}?"
    r"(this protocol tab|living protocol)")
# The same item answered No, with the reason given.
HONEST_NO = re.compile(
    r"(?is)protocol registered before commencement.{0,400}?\bno\b.{0,300}?"
    r"(retrospective|not prospectively registered)")
# A bare count of unverified numbers, with no identifier exposed.
BARE_COUNT = re.compile(
    r"(?i)(\d{1,4})\s*number\(?s?\)?\s*on this page marked\s*UNVERIFIED")
# A named identifier flagged unverified -- checkable by a reader.
NAMED_UNVERIFIED = re.compile(r"(?i)UNVERIFIED.{0,200}?\b(NCT\d{8})\b")


def git(*a):
    return subprocess.run(["git"] + list(a), cwd=REPO, capture_output=True)


def text(p):
    r = git("show", "%s:%s" % (REF, p))
    return "" if r.returncode else r.stdout.decode("utf-8", "replace")


def main():
    sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8", errors="replace")
    if not os.path.isfile(CLASSES):
        sys.exit("REFUSED: %s is missing. Run classify_five_buckets first -- this script "
                 "reports per family and cannot invent the families."
                 % os.path.relpath(CLASSES, REPO))
    cls = json.load(io.open(CLASSES, encoding="utf-8"))
    names = sorted(set(cls.get("legacy", []) + cls.get("audit-first", [])
                       + cls.get("unclassifiable", [])))

    ctrl = text("BARIATRIC_RYGB_VS_SG_REVIEW.html")
    ctrl_bare = bool(BARE_COUNT.search(ctrl)) if ctrl else None
    sample_full = next((n for n in names if "_AUTO_FULL" in n and HONEST_NO.search(text(n))),
                       None)
    require_controls(
        "registration_claims_by_family",
        ("BARIATRIC_RYGB_VS_SG carries a bare UNVERIFIED count with no identifier "
         "(reported by a census lane reading the page)", bool(ctrl_bare), True),
        ("an *_AUTO_FULL page answering 'No -- retrospective, not prospectively registered' "
         "is NOT counted as self-certifying (found %s)" % sample_full,
         bool(sample_full) and bool(SELF_CERT.search(text(sample_full))), True))

    fam = collections.Counter()
    self_cert, honest, bare, named, both = [], [], [], [], []
    bare_totals = {}
    read = 0
    for n in names:
        t = text(n)
        if not t:
            continue
        read += 1
        f = "AUTO_FULL" if "_AUTO_FULL" in n else (
            "AUTO_2" if "_AUTO_2" in n else "NMA/broad")
        fam[f] += 1
        sc = bool(SELF_CERT.search(t))
        hn = bool(HONEST_NO.search(t))
        bc = BARE_COUNT.search(t)
        nu = NAMED_UNVERIFIED.search(t)
        if sc:
            self_cert.append((n, f))
        if hn:
            honest.append((n, f))
        if bc:
            bare.append(n)
            bare_totals[n] = int(bc.group(1))
        if nu:
            named.append(n)
        if bc and nu:
            both.append(n)

    print("")
    print("REGISTRATION CLAIMS BY FAMILY, %d non-SSOT reachable page(s) at %s" % (read, REF))
    print("")
    print("   family distribution:")
    for k, v in fam.most_common():
        print("      %-24s %5d" % (k, v))
    print("")
    print("   THE DEFECT -- AMSTAR item 2 answered YES, self-certified via the page's own")
    print("   protocol tab, no external registration id:      %5d" % len(self_cert))
    for n, f in self_cert[:10]:
        print("      %-52s %s" % (n[:52], f))
    print("")
    print("   NOT A DEFECT -- answered NO, reason given ('retrospective public protocol")
    print("   pack, not prospectively registered'):           %5d" % len(honest))
    print("      This family volunteered a correct answer where a false Yes would never")
    print("      have been checked. It is left alone, and that is a finding, not a courtesy.")
    print("")
    print("   UNVERIFIED IDENTIFIERS, two shapes:")
    print("      a NAMED unresolvable id a reader can look up      %5d" % len(named))
    print("      a BARE COUNT with no identifier exposed           %5d" % len(bare))
    print("      pages carrying BOTH                               %5d" % len(both))
    if bare_totals:
        top = sorted(bare_totals.items(), key=lambda kv: -kv[1])[:8]
        print("      largest bare counts:")
        for n, c in top:
            print("         %-50s %4d numbers" % (n[:50], c))
    print("")
    if both:
        print("   THE DISTINCTION IS WEAKER THAN IT LOOKS ON %d PAGE(S): they carry a bare" % len(both))
        print("   count AND named ids, so a reader does have something to check there.")
    else:
        print("   THE DISTINCTION HOLDS: no page carries both, so a bare-count page exposes")
        print("   nothing a reader can look up, which is the argument for separating them.")

    json.dump({"ref": REF, "read": read, "family": dict(fam),
               "self_certifying": [n for n, _f in self_cert],
               "honest_no": [n for n, _f in honest],
               "bare_count": bare_totals, "named_unverified": named, "both": both},
              io.open(OUT, "w", encoding="utf-8"), indent=1)


if __name__ == "__main__":
    main()
