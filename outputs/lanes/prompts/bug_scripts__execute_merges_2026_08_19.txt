#!/usr/bin/env python3
"""EXECUTE THE AUTHORISED MERGES -- and prove the survivor holds the union.

Mahmood authorised the merges. A merge DISCARDS AN OBJECT, and that is the one direction this
project's guards do not cover: every other gate here defends against a wrong number, and this
one has to defend against a MISSING one. So the rules are mechanical and each is asserted:

  NOTHING IS DELETED. A merged-away topic is RETIRED, not removed. Its object stays on disk,
  rewritten as a TOMBSTONE that records what it was, which topic absorbed it, when and why, and
  its full former content is carried into the survivor verbatim.

  THE SURVIVOR HOLDS THE UNION, AND IT IS PROVEN. Every LEAF of the retired object must be
  findable in the survivor afterwards. Not eyeballed -- counted, with the missing paths named
  and the merge REFUSED if any is absent.

  A READER WITH A LINK MUST NOT HIT A 404. The retired page is replaced by a tombstone page
  that states plainly that the question is answered at the surviving URL, and links to it.

  AND A MERGE IS REFUSED IF THE SURVIVOR HAS NO PAGE. Three of the twelve clusters have a
  page-bearing topic being retired in favour of a richer object that has never been built. A
  tombstone pointing at a URL that does not exist is worse than the duplicate it replaces, so
  those merges are DEFERRED and reported, not executed with a broken target.

USAGE:  python scripts/execute_merges_2026_08_19.py            # dry run, changes nothing
        python scripts/execute_merges_2026_08_19.py --apply

def leaves(o, p=""):
    if isinstance(o, dict):
        for k, v in o.items():
            for x in leaves(v, p + "/" + str(k)):
                yield x
    elif isinstance(o, list):
        for i, v in enumerate(o):
            for x in leaves(v, "%s[%d]" % (p, i)):
                yield x
    else:
        yield (p, o)


def obj_path(topic):
    return os.path.join(REPO, "ssot", topic, topic + ".json")


def load(topic):
    with io.open(obj_path(topic), "r", encoding="utf-8") as fh:
        return json.load(fh)


def page_for(topic, pmap):
    for page, rel in pmap.items():
        if rel.rsplit("/", 1)[-1] == topic + ".json":
            return page
    return None


def prove_union(survivor_after, retired_obj, retired_topic):
    """Every leaf of the retired object must be findable in the survivor. Names what is not.

    THE COMPARISON IS ON (path-suffix, value) PAIRS, not on absolute paths, because the retired
    content is carried under `absorbed_topics/<topic>/...` and its paths therefore change. What
    must not change is the CONTENT: every (leaf name, value) the retired object held has to be
    somewhere in the survivor.
    """
    want = list(leaves(retired_obj))
    have = set()
    for p, v in leaves(survivor_after):
        have.add((p.rsplit("/", 1)[-1], json.dumps(v, sort_keys=True)))
    missing = []
    for p, v in want:
        key = (p.rsplit("/", 1)[-1], json.dumps(v, sort_keys=True))
        if key not in have:
            missing.append(p)
    return missing, len(want)

def selftest():
    fails = []

    def ck(name, got, want):
        ok = got == want
        print("  %-70s %s  %r" % (name, "ok" if ok else "FAIL", got))
        if not ok:
            fails.append(name)

    print("1. THE UNION PROOF FINDS A MISSING VALUE, which is the whole point:")
    retired = {"a": 1, "b": {"c": "kept"}, "d": "LOST"}
    surv_ok = {"absorbed_topics": {"x": {"THE_WHOLE_OBJECT_VERBATIM": retired}}}
    surv_bad = {"absorbed_topics": {"x": {"THE_WHOLE_OBJECT_VERBATIM": {"a": 1, "b": {"c": "kept"}}}}}
    ck("a survivor carrying the object verbatim proves the union",
       prove_union(surv_ok, retired, "x")[0], [])
    ck("a survivor missing one value NAMES it", prove_union(surv_bad, retired, "x")[0], ["/d"])

    print("\n2. AND IT COMPARES CONTENT, NOT ABSOLUTE PATHS -- the retired content moves:")
    ck("the path changed and the proof still holds",
       prove_union({"anywhere": {"else": {"d": "LOST"}}}, {"d": "LOST"}, "x")[0], [])

    print("\n3. A VALUE THAT MERELY LOOKS SIMILAR IS NOT A MATCH:")
    ck("same leaf name, different value -> missing",
       prove_union({"q": {"d": "different"}}, {"d": "LOST"}, "x")[0], ["/d"])

    print("\n4. THE DRY RUN CHANGES NOTHING -- run it and confirm the objects are untouched:")
    import hashlib
    with io.open(ADJ, "r", encoding="utf-8") as fh:
        adj = json.load(fh)
    tops = [m for r in adj["rows"] for m in r["cluster"] if os.path.exists(obj_path(m))]
    before = {t: hashlib.md5(io.open(obj_path(t), "rb").read()).hexdigest() for t in tops}
    run(apply_changes=False)
    after = {t: hashlib.md5(io.open(obj_path(t), "rb").read()).hexdigest() for t in tops}
    ck("no object changed during the dry run", before == after, True)

    print("\n%s" % ("SELFTEST FAILED: %s" % fails if fails else "SELFTEST PASSED"))
    return 1 if fails else 0
