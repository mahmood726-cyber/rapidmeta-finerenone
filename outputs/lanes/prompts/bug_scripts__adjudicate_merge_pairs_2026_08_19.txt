#!/usr/bin/env python3
"""ADJUDICATE THE 15 IDENTICAL-TRIAL-SET PAIRS -- evidence first, direction second.

Mahmood authorised the merges. This file does NOT merge. It assembles, for every pair, the
evidence a merge direction has to be argued from, and it applies the three rules that were set
before it ran:

  1. IS IT ACTUALLY A DUPLICATE? An identical trial set is a QUESTION, never a verdict. Two
     topics may ask legitimately different questions of the same evidence -- that is P21 in
     reverse -- and collapsing them silently would discard a QUESTION rather than a duplicate.
     A pair whose questions differ materially is reported as NOT A MERGE.
  2. MERGE IN THE DIRECTION OF EVIDENCE, NOT OF TIDINESS. Where one member is a subset of the
     other, the superset survives. Where the sets are identical and one object is materially
     richer, the richer survives.
  3. WHERE NEITHER IS RICHER, THE BETTER QUESTION SURVIVES and the other's question is recorded
     on it as an alternative reading.

AND THE PAIRS ARE NOT INDEPENDENT. `omecamtiv-heartfail`, `omecamtiv-hf` and `omecamtiv-hfref`
hold the same single trial and appear as THREE pairs of one three-way cluster. `mavacamten-ohcm`
and `mavacamten-ohcm-review` are identical to each other AND both subsets of
`mavacamten-hcm-review`. `sacubitril-heartfail` and `sacubitril-valsartan-hf` are identical AND
both subsets of `arni-hfref`. TREATING THOSE AS PAIRWISE MERGES WOULD RETIRE A TOPIC INTO
ANOTHER TOPIC THAT IS ITSELF ABOUT TO BE RETIRED, so clusters are resolved as clusters.

USAGE:  python scripts/adjudicate_merge_pairs_2026_08_19.py
        python scripts/adjudicate_merge_pairs_2026_08_19.py --selftest
"""
import io
import json
import os
import sys

REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
EV = os.path.join(REPO, "evidence", "2026-08-19-batch1")
AUDIT = os.path.join(EV, "corpus_duplication.json")
DEST = os.path.join(EV, "merge_adjudication.json")


def obj_path(topic):
    return os.path.join(REPO, "ssot", topic, topic + ".json")


def load(topic):
    p = obj_path(topic)
    if not os.path.exists(p):
        return None
    try:
        with io.open(p, "r", encoding="utf-8") as fh:
            return json.load(fh)
    except (OSError, ValueError):
        return None


def leaves(o, p=""):
    if isinstance(o, dict):
        for k, v in o.items():
            for x in leaves(v, p + "/" + str(k)):
                yield x
    elif isinstance(o, list):
        for i, v in enumerate(o):
            for x in leaves(v, "%s[%d]" % (p, i)):
                yield x
    else:
        yield (p, o)


def profile(topic):
    """Everything a merge direction can be argued from. Counted, never eyeballed."""
    o = load(topic)
    if o is None:
        return {"topic": topic, "state": "NO_OBJECT_ON_DISK"}
    lv = dict(leaves(o))
    outs = ((o.get("results") or {}).get("by_outcome")) or {}
    trials = ((o.get("inputs") or {}).get("trials")) or []
    pooled = []
    for oid, blk in outs.items():
        if not isinstance(blk, dict):
            continue
        p = blk.get("pooled") or {}
        pooled.append({"outcome": oid, "measure": p.get("measure"), "point": p.get("point"),
                       "withdrawn": bool(p.get("withdrawn")), "k": blk.get("k")})
    return {
        "topic": topic, "state": "READ",
        "title": (o.get("title") or "")[:200],
        "question": (o.get("question") or "")[:600],
        "leaf_count": len(lv),
        "bytes": os.path.getsize(obj_path(topic)),
        "n_trials": len(trials),
        "trials": sorted(t.get("nct") or t.get("id") or "?" for t in trials),
        "n_outcomes": len(outs),
        "outcomes": pooled,
        "has_build_stamp": bool(o.get("build_stamp")),
        "standard_version": ((o.get("build_stamp") or {}).get("page_standard_version")
                             or (o.get("build_stamp") or {}).get("standard_version")),
        "has_search": bool(o.get("search")),
        "has_k_cascade": bool(o.get("k_cascade")),
        "has_prisma": bool(o.get("prisma_flow")),
        "has_screening": bool(o.get("screening")),
        "n_corrections": len(o.get("claims_corrected") or []),
        "has_superseded_blocks": sum(1 for k in lv if "supersed" in k.lower()),
        "has_restated_blocks": sum(1 for k in lv if "restat" in k.lower()),
        "n_provenance_blocks": sum(1 for k in lv if "provenance" in k.lower()),
        "n_limitations": sum(1 for k in lv if "limitation" in k.lower()
                             or "known_limitation" in k.lower()),
        "build_mode": o.get("build_mode"),
    }


# RICHNESS IS A COMPOSITE OF THINGS THAT WOULD BE LOST, not of file size. A bigger file can be
# bigger because it embeds a longer prose block; what matters for a merge is whether it holds
# STRUCTURE the other does not.
RICHNESS_KEYS = ("has_build_stamp", "has_search", "has_k_cascade", "has_prisma",
                 "has_screening")
COUNT_KEYS = ("n_outcomes", "n_corrections", "has_superseded_blocks", "has_restated_blocks",
              "n_provenance_blocks", "n_limitations", "leaf_count")


def richness(p):
    if p.get("state") != "READ":
        return -1
    return (sum(1 for k in RICHNESS_KEYS if p.get(k)) * 1000
            + p.get("n_outcomes", 0) * 100
            + p.get("n_corrections", 0) * 50
            + p.get("has_superseded_blocks", 0) * 10
            + p.get("n_provenance_blocks", 0)
            + p.get("leaf_count", 0) // 100)


# THE AUTO-BUILD TEMPLATE. 37 of the corpus's 144 questions end in this exact clause, and none
# of them is a question -- each is the object's TITLE with a stock suffix appended. Comparing a
# templated title against a real question measures the template.
TEMPLATE_TAIL = "on each trial's own registered primary outcome?"


def is_templated(q):
    return (q or "").strip().lower().endswith(TEMPLATE_TAIL)


def questions_differ(a, b):
    """Do the two topics ask MATERIALLY different questions? Conservative on purpose.

    A merge that collapses two different questions destroys evidence in the direction this
    project has no guard for. So the test errs toward REPORTING a difference: identical or
    empty questions are the only confident duplicate.

    AND A TEMPLATED TITLE IS AN ABSENT QUESTION, NOT A DIFFERENT ONE. The first run of this file
    flagged `lenacapavir-prep` against `lenacapavir-prep-review` at 6% overlap and
    `cryptococcal-meningitis` against its `-africa` stub at 13%, and neither pair asks two
    questions: one member of each is an auto-built stub whose `question` field is its own title
    with `on each trial's own registered primary outcome?` appended. Treating that as a rival
    question would have blocked two merges for the wrong reason -- and, worse, would have
    reported a REAL finding ("these are two questions") that was an artefact of a template.
    """
    qa, qb = (a.get("question") or "").strip(), (b.get("question") or "").strip()
    if not qa and not qb:
        return False, "neither object records a question"
    if not qa or not qb:
        return False, "only one object records a question, so there is nothing to differ"
    if is_templated(qa) and is_templated(qb):
        return False, "BOTH questions are the auto-build template, so neither states a question"
    if is_templated(qa) or is_templated(qb):
        which = a.get("topic") if is_templated(qa) else b.get("topic")
        return False, ("%s carries the AUTO-BUILD TEMPLATE, not a question -- its `question` "
                       "field is its own title with a stock suffix. An absent question is not a "
                       "different question." % which)
    if qa == qb:
        return False, "the two questions are byte-identical"
    # Token overlap on the content words.
    stop = set("does do is are the a an of in for with and or on at to from by as this that "
               "than versus vs compared comparison patients people adults".split())
    ta = {w for w in "".join(c.lower() if c.isalnum() else " " for c in qa).split()
          if w not in stop and len(w) > 2}
    tb = {w for w in "".join(c.lower() if c.isalnum() else " " for c in qb).split()
          if w not in stop and len(w) > 2}
    if not ta or not tb:
        return False, "one question carries no content words"
    j = len(ta & tb) / float(len(ta | tb))
    if j >= 0.6:
        return False, "questions overlap %.0f%% on content words" % (100 * j)
    return True, ("questions overlap only %.0f%% on content words -- REPORT, do not merge, "
                  "until a human confirms they are one question" % (100 * j))


def main():
    with io.open(AUDIT, "r", encoding="utf-8") as fh:
        audit = json.load(fh)
    pairs = audit["identical_trial_sets"]
    subs = audit["subset_relations"]
    superset_of = {}
    for s in subs:
        superset_of.setdefault(s["subset"], []).append(s["superset"])

    # CLUSTERS, not pairs. Union-find over the identical-set relation.
    parent = {}

    def find(x):
        parent.setdefault(x, x)
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[ra] = rb

    for p in pairs:
        union(p["topics"][0], p["topics"][1])
    clusters = {}
    for t in parent:
        clusters.setdefault(find(t), []).append(t)

    rows = []
    for root, members in sorted(clusters.items(), key=lambda kv: sorted(kv[1])[0]):
        members = sorted(members)
        profs = {m: profile(m) for m in members}
        diffs = []
        for i in range(len(members)):
            for j in range(i + 1, len(members)):
                d, why = questions_differ(profs[members[i]], profs[members[j]])
                diffs.append({"a": members[i], "b": members[j], "questions_differ": d,
                              "why": why})
        any_diff = any(d["questions_differ"] for d in diffs)

        # IS THE PAIR ACTUALLY AN IDENTICAL SET? RE-CHECKED HERE, AGAINST EVERY TRIAL.
        #
        # `corpus_duplication.json` compared trial sets keyed on NCT ids. A trial registered
        # anywhere else -- ANZCTR, ISRCTN, EudraCT -- is INVISIBLE to that comparison, so a
        # topic holding three trials of which one has an NCT looks identical to a topic holding
        # that one trial. `cryptococcal-meningitis` holds ACTA, AMBITION-cm and NCT01075152;
        # `cryptococcal-meningitis-africa` holds NCT01075152 alone. THE AUDIT REPORTED THEM AS
        # AN IDENTICAL SET AND THEY ARE A SUBSET RELATION.
        #
        # Same class as the P19 check that could not carry an ANZCTR trial, found hours earlier
        # on `colchicine-cvd-coronary`: A COMPARISON KEYED ON ONE REGISTRY'S IDENTIFIER CANNOT
        # SEE TRIALS FROM ANOTHER. Merging on an unverified pair list would retire a topic into
        # one that does not contain it.
        sets = {m: set(profs[m].get("trials") or []) for m in members
                if profs[m].get("state") == "READ"}
        sizes = {m: len(s) for m, s in sets.items()}
        truly_identical = len({frozenset(s) for s in sets.values()}) == 1
        set_note = None
        if not truly_identical:
            biggest = max(sets, key=lambda m: len(sets[m]))
            others_are_subsets = all(sets[m] <= sets[biggest] for m in sets)
            set_note = {
                "audit_said": "identical trial sets",
                "recheck_says": ("NOT IDENTICAL over ALL trials: %s" % sizes),
                "why_the_audit_missed_it": ("it compared sets keyed on NCT ids, and a trial "
                                            "registered on another registry is invisible to "
                                            "that comparison"),
                "relation": ("SUBSET -- every other member is contained in %s" % biggest
                             if others_are_subsets else "PARTIAL OVERLAP"),
            }
        # Is any member a subset of a topic OUTSIDE the cluster? Then the cluster's survivor is
        # itself absorbed and the merge must not be taken pairwise.
        outside = {}
        for m in members:
            up = [s for s in superset_of.get(m, []) if s not in members]
            if up:
                outside[m] = up
        ranked = sorted(members, key=lambda m: -richness(profs[m]))
        survivor = ranked[0]
        retire = [m for m in members if m != survivor]
        verdict = "MERGE"
        note = ("survivor is the richest object in the cluster by structure held, not by file "
                "size")
        if not truly_identical:
            verdict = "NOT_AN_IDENTICAL_SET_REPORT_ONLY"
            note = ("THE AUDIT'S PAIRING IS WRONG FOR THIS CLUSTER and no merge is taken on it. "
                    "%s. A merge authorised for identical sets must not be executed on a pair "
                    "that is not one." % set_note["recheck_says"])
        elif any_diff:
            verdict = "NOT_A_MERGE_REPORT_ONLY"
            note = ("at least one pair in this cluster asks materially different questions of "
                    "the same evidence. THAT IS P21 IN REVERSE and collapsing it would discard "
                    "a question rather than a duplicate.")
        elif outside:
            verdict = "MERGE_BUT_THE_CLUSTER_IS_ITSELF_A_SUBSET"
            note = ("every member is a subset of %s. Merging inside the cluster is correct and "
                    "INCOMPLETE: the survivor should then be considered against its superset, "
                    "which is a SECOND decision and is not taken here."
                    % sorted({s for v in outside.values() for s in v}))
        if any(profs[m]["state"] != "READ" for m in members):
            verdict = "NOT_ASSESSABLE"
            note = "at least one member has no readable object on disk"
        rows.append({
            "cluster": members, "verdict": verdict, "survivor": survivor,
            "trial_sets_rechecked_over_ALL_trials": {m: sorted(sets.get(m, ())) for m in members},
            "truly_identical": truly_identical, "set_recheck": set_note,
            "retire": retire, "note": note,
            "richness": {m: richness(profs[m]) for m in members},
            "question_comparison": diffs,
            "subset_of_topics_outside_the_cluster": outside,
            "profiles": profs,
        })

    out = {
        "adjudicated_utc": "2026-08-19",
        "authority": ("Mahmood authorised the merges. The direction of each one is argued from "
                      "evidence here and no object is written by this file."),
        "rules_applied_and_set_before_this_ran": [
            "An identical trial set is a QUESTION, never a verdict. A pair whose questions "
            "differ materially is REPORTED, not merged.",
            "Where one member is a subset of another topic, the superset survives.",
            "Where sets are identical, the object holding more STRUCTURE survives -- search, "
            "cascade, PRISMA, screening, outcomes, corrections, supersede chains, provenance. "
            "NOT file size.",
            "Where neither is richer, the better question survives and the other's question is "
            "recorded on it as an alternative reading.",
        ],
        "clusters": len(rows),
        "pairs_reported_by_the_audit": len(pairs),
        "why_those_two_numbers_differ": (
            "The 15 pairs are not 15 independent merges. The three omecamtiv topics hold the "
            "same single trial and appear as THREE pairs of ONE three-way cluster. Treating "
            "them pairwise would retire a topic into a topic that is itself being retired."),
        "rows": rows,
    }
    with io.open(DEST, "w", encoding="utf-8") as fh:
        fh.write(json.dumps(out, indent=1, ensure_ascii=False))

    print("%d identical-set PAIRS resolve into %d CLUSTERS\n" % (len(pairs), len(rows)))
    for r in rows:
        print("%-38s %s" % (" + ".join(r["cluster"])[:38], r["verdict"]))
        print("     survivor  %s   retire  %s" % (r["survivor"], r["retire"]))
        for m in r["cluster"]:
            p = r["profiles"][m]
            if p["state"] != "READ":
                print("       %-40s %s" % (m, p["state"]))
                continue
            print("       %-40s richness %6d  leaves %5d  outcomes %d  k=%d  stamp=%s"
                  % (m, r["richness"][m], p["leaf_count"], p["n_outcomes"], p["n_trials"],
                     p["standard_version"] or "-"))
        for d in r["question_comparison"]:
            if d["questions_differ"]:
                print("       QUESTIONS DIFFER: %s" % d["why"])
        if r["subset_of_topics_outside_the_cluster"]:
            print("       subset of: %s" % r["subset_of_topics_outside_the_cluster"])
        print("")
    print("wrote %s" % os.path.relpath(DEST, REPO))
    return 0


def selftest():
    fails = []

    def ck(name, got, want):
        ok = got == want
        print("  %-70s %s  %r" % (name, "ok" if ok else "FAIL", got))
        if not ok:
            fails.append(name)

    print("1. THE QUESTION TEST ERRS TOWARD REPORTING A DIFFERENCE:")
    same = {"question": "Does X reduce Y in Z?"}
    ck("identical questions are not a difference", questions_differ(same, same)[0], False)
    ck("two unrelated questions ARE a difference",
       questions_differ({"question": "Does colchicine reduce stroke recurrence?"},
                        {"question": "Does bosentan improve walk distance in children?"})[0],
       True)
    ck("a missing question on one side is not a difference",
       questions_differ({"question": "Does X reduce Y?"}, {"question": ""})[0], False)

    print("\n2. RICHNESS IS STRUCTURE HELD, NOT FILE SIZE:")
    lean = {"state": "READ", "leaf_count": 20000, "n_outcomes": 1}
    rich = {"state": "READ", "leaf_count": 400, "n_outcomes": 2, "has_build_stamp": True,
            "has_search": True, "has_k_cascade": True, "has_prisma": True,
            "has_screening": True}
    ck("a small structured object outranks a large unstructured one",
       richness(rich) > richness(lean), True)

    print("\n3. AN UNREADABLE OBJECT IS NOT_ASSESSABLE, never the loser by default:")
    ck("no object on disk ranks -1", richness({"state": "NO_OBJECT_ON_DISK"}), -1)

    print("\n4. THE LIVE ADJUDICATION RUNS AND CLUSTERS THE OMECAMTIV TRIPLE AS ONE:")
    rc = main()
    with io.open(DEST, "r", encoding="utf-8") as fh:
        out = json.load(fh)
    om = [r for r in out["rows"] if any("omecamtiv" in m for m in r["cluster"])]
    ck("the three omecamtiv topics form ONE cluster", len(om), 1)
    if om:
        ck("and it has three members", len(om[0]["cluster"]), 3)
    ck("15 pairs resolve into fewer clusters", out["clusters"] < 15, True)

    print("\n%s" % ("SELFTEST FAILED: %s" % fails if fails else "SELFTEST PASSED"))
    return 1 if fails else 0
