def woolf(tE, tN, cE, cN):
    """log OR and its Woolf variance, matching the convention already in the blob."""
    a, b, c, d = tE, tN - tE, cE, cN - cE
    y = math.log((a * d) / (b * c))
    v = 1.0 / a + 1.0 / b + 1.0 / c + 1.0 / d
    return y, v


def mh_or(rows):
    """Mantel-Haenszel OR with Robins-Breslow-Greenland SE."""
    R = S = sPR = sPSQR = sQS = 0.0
    for tE, tN, cE, cN in rows:
        a, b, c, d = tE, tN - tE, cE, cN - cE
        N = tN + cN
        Ri, Si = a * d / N, b * c / N
        P, Q = (a + d) / N, (b + c) / N
        R += Ri
        S += Si
        sPR += P * Ri
        sPSQR += P * Si + Q * Ri
        sQS += Q * Si
    OR = R / S
    var = sPR / (2 * R * R) + sPSQR / (2 * R * S) + sQS / (2 * S * S)
    se = math.sqrt(var)
    return OR, math.exp(math.log(OR) - 1.96 * se), math.exp(math.log(OR) + 1.96 * se), se

def patch_stub(path, OR, lo, hi):
    """The stub carried neither verdict surface. Give it both, agreeing with the full app."""
    s = open(path, encoding="utf-8").read()
    assert "__verdict" not in s, "stub already carries a verdict surface"
    banner = (
        '<div id="rapidmeta-integrity-badge" role="status" '
        'style="background:#7c2d12;color:#fff;padding:12px 20px;font-family:system-ui,sans-serif;'
        'font-size:13.5px;border-bottom:3px solid #431407;line-height:1.55;">'
        '<strong style="font-size:14px;letter-spacing:0.04em;">VERDICT: UNCERTAIN '
        '&mdash; CORRECTED, AND THE CONCLUSION REVERSED</strong>'
        '<div style="margin-top:6px;font-size:12.5px;">This page previously reported a '
        'significant <em>benefit</em> of apixaban (OR 0.850). Corrected and re-sourced, '
        f'apixaban roughly <strong>doubles</strong> ISTH major/CRNM bleeding: '
        f'<strong>OR {OR:.3f} ({lo:.3f}&ndash;{hi:.3f})</strong>, k=2. '
        'See the full review for the ledger, the quarantined trials and the stated limits. '
        'AMSTAR-2 confidence: <strong>CRITICALLY LOW</strong>.</div></div>\n'
    )
    verdict_js = (
        '<script>window.__verdict = {"verdict":"UNCERTAIN","counts":{"n_trials_seen":2,'
        '"n_trials_quarantined":2,"p0_total":0},"reasons":["Redirect stub. Verdict mirrors '
        'APIXABAN_ACS_AUTO_FULL_REVIEW.html: pooled k=2, 2 trials quarantined, prior pooled '
        'estimate superseded after manufactured counts and inverted arms were corrected."],'
        '"p0_total":0,"pooled":{"estimator":"Mantel-Haenszel OR, Robins-Breslow-Greenland SE",'
        f'"k":2,"or":{OR:.4f},"ci":[{lo:.4f},{hi:.4f}]}}}};</script>\n'
    )
    m = re.search(r"<body[^>]*>", s)
    if m:
        s = s[:m.end()] + "\n" + banner + verdict_js + s[m.end():]
    else:
        s = s.replace("</head>", "</head>\n" + verdict_js, 1)
        s = s.replace("<body>", "<body>\n" + banner, 1)
    open(path, "w", encoding="utf-8", newline="").write(s)
    print(f"{path}: badge + __verdict added (both surfaces now present and agreeing)")
