def load(topic):
    p = os.path.join(REPO, "ssot", topic, topic + ".json")
    with io.open(p, encoding="utf-8") as fh:
        return p, json.load(fh)


def save(path, obj):
    with io.open(path, "w", encoding="utf-8") as fh:
        fh.write(json.dumps(obj, indent=1, ensure_ascii=True))


def supersede(container, key, new_value, why):
    """Replace container[key], keeping the old value under a superseded key beside it."""
    old = container.get(key)
    if old is not None:
        container[key + "_superseded_2026_08_19"] = {"was": old, "superseded_because": why}
    container[key] = new_value
