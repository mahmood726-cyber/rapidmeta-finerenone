def main():
    with io.open(OBJ, encoding="utf-8") as fh:
        obj = json.load(fh)

    rm = obj["removed_citations"]
    note = rm["disclosure_note"]
    for k in ("total_cited", "removed", "retained", "contributing_to_this_outcome"):
        note = note.replace("{rm.%s}" % k, str(rm[k]))
    rm["disclosure_note"] = note + (
        " NOTE ADDED 2026-08-19: these counts describe citations retained from the SOURCE APP. "
        "The two trials recovered by screening on that date (NCT02289963, NCT02585778) were "
        "never in that citation list -- they were found in the registry -- so 6 remains "
        "correct here and must not be read as the number of trials in the pool, which is 8.")

    scr["eligibility_provenance"] = PROVENANCE

    with io.open(OBJ, "w", encoding="utf-8", newline="") as fh:
        fh.write(json.dumps(obj, indent=1, ensure_ascii=False) + "\n")
