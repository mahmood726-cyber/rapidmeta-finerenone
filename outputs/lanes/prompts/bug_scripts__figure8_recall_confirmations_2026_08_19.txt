def reg(nct):

    ps = s.get("protocolSection") or {}
    return {
        "nct": nct,
        "acronym": (ps.get("identificationModule") or {}).get("acronym"),
        "conditions_VERBATIM": (ps.get("conditionsModule") or {}).get("conditions"),
