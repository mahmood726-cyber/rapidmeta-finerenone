"""Fields written onto objects tonight, and whether any consumer reads them.

REGISTRY CLASS 65 SAYS: a correct object, a correct refusal, and no consumer obliged to read
it. THE AUTHOR OF THAT ENTRY BREACHED IT WITHIN THE HOUR -- a complete risk-of-bias
assessment for both EMPEROR results was written to
`risk_of_bias.by_outcome["primary/NCT03057977"]`, a flat key nothing reads, while
`scripts/p46_queue.py` went on reporting the limb REFUSED. The assessment was correct,
present, and invisible.

    A CORRECT ARTEFACT NOBODY READS IS INDISTINGUISHABLE FROM AN ABSENT ONE.

So this asks the question of everything written tonight rather than waiting for the next
scorer to disagree with a field. For each key added to the objects on 2026-08-20, it greps
the renderers, the projectors and the scorers for that key name.

WHAT A "NO" MEANS, AND WHAT IT DOES NOT. Some of these fields are written FOR A HUMAN
READING THE OBJECT -- a finding recorded where the next person to open the file will meet
it. That is a legitimate purpose and it is not the same as an oversight. What the sweep
gives is the list; which are deliberate is a judgement, and the judgement is written beside
each one rather than assumed.

THE SWEEP CANNOT SEE DYNAMIC ACCESS. A renderer that walks every key of a block -- `for k, v
in blk.items()` -- reads these without naming them. Such a walk is reported separately,
because "no consumer names this key" and "no consumer reads this key" are different claims
and only the first is measurable here.
"""
import io
import os
import re
import sys
import json
import glob

REPO = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from instrument_controls import require_controls

CONSUMERS = ["ssot/build_tabbed.py", "ssot/build_app_v2.py", "ssot/paper_projector.py",
             "ssot/projectors.py", "ssot/projectors2.py", "ssot/build_to_standard.py",
             "ssot/validate_v2.py", "ssot/assessment.py", "ssot/preconditions.py"]
SELF = os.path.abspath(__file__)
# THIS FILE MUST NOT COUNT AS A CONSUMER OF THE KEYS IT NAMES. The first version globbed
# audit_*.py into the scorer list and so included ITSELF -- every key in WRITTEN_TONIGHT
# appears in this source, so every one would have come back "read", and the sweep would
# have reported a clean corpus BECAUSE IT WAS READING ITS OWN TABLE. Caught by the negative
# control on the first run: a key that exists nowhere came back as read.
SCORERS = [p for p in (glob.glob(os.path.join(REPO, "scripts", "p46_queue.py"))
                       + glob.glob(os.path.join(REPO, "scripts", "audit_*.py"))
                       + glob.glob(os.path.join(REPO, "scripts", "lint_*.py")))
           if os.path.abspath(p) != SELF]

# Keys written onto objects on 2026-08-20, with what each is FOR.
WRITTEN_TONIGHT = [
    ("THE_POOL_IS_REFERRED_2026_08_20",
     "attr-pn, sglt2-mace: the referral and its reason"),
    ("estimand_established_does_not_cover_the_contrast_2026_08_20",
     "156 outcome blocks: the P48 gap named beside the flag"),
    ("registration_read_2026_08_20",
     "trials: the registration read verbatim, with the arm reversal named"),
    ("arm_roles_contradicted_by_this_object_2026_08_20",
     "8 trial rows: the label/name/role contradictions"),
    ("estimator_note_2026_08_20",
     "10 pools: why the estimator moved to REML"),
    ("house_rule_interval_2026_08_18",
     "pools: the floored Hartung-Knapp sensitivity interval"),
    ("r_output",
     "pools: the model output quoted verbatim -- P46's fourth criterion"),
    ("risk_of_bias",
     "empagliflozin: the per-result assessment written tonight"),
    ("display_change_announced",
     "the log of what changed and why"),
]

DYNAMIC_WALK = re.compile(r"\.items\(\)|for\s+\w+\s*,\s*\w+\s+in\s+\w+\.items")


def reads(name, paths):
    hits = []
    for p in paths:
        try:
            src = io.open(p, encoding="utf-8", errors="replace").read()
        except OSError:
            continue
        if name in src:
            hits.append(os.path.relpath(p, REPO).replace("\\", "/"))
    return hits


def main():
    consumer_paths = [os.path.join(REPO, c) for c in CONSUMERS]
    require_controls(
        "audit_written_tonight_is_read_by_something",
        positive=("`r_output` -- P46's verbatim-output criterion, which the paper "
                  "projector demonstrably names",
                  bool(reads("r_output", consumer_paths + SCORERS)), True),
        negative=("a key nothing could possibly read",
                  bool(reads("__no_such_key_written_anywhere__",
                             consumer_paths + SCORERS)), True))

    dyn = [os.path.relpath(p, REPO).replace("\\", "/") for p in consumer_paths
           if os.path.exists(p) and DYNAMIC_WALK.search(
               io.open(p, encoding="utf-8", errors="replace").read())]

    print("")
    print("%-58s %-22s %s" % ("key written 2026-08-20", "named by a RENDERER",
                              "named by a SCORER"))
    print("-" * 110)
    unread = []
    for name, what in WRITTEN_TONIGHT:
        r = reads(name, consumer_paths)
        s = reads(name, SCORERS)
        print("%-58s %-22s %s" % (name[:58], "yes (%d)" % len(r) if r else "NO",
                                  "yes (%d)" % len(s) if s else "NO"))
        print("        %s" % what)
        if not r and not s:
            unread.append((name, what))

    print("")
    print("NAMED BY NOTHING: %d of %d" % (len(unread), len(WRITTEN_TONIGHT)))
    for name, what in unread:
        print("    %s" % name)
        print("        %s" % what)

    print("")
    print("%d of the %d consumer files walk keys dynamically (`.items()`), so a field they"
          % (len(dyn), len(CONSUMERS)))
    print("do not NAME may still reach a page: %s" % ", ".join(os.path.basename(d)
                                                               for d in dyn))
    print("`no consumer names this key` and `no consumer reads this key` are DIFFERENT")
    print("claims and only the first is measured here.")
    print("")
    print("AND A KEY NAMED BY NOTHING IS NOT AUTOMATICALLY A DEFECT. Several of these are")
    print("written FOR A HUMAN OPENING THE OBJECT -- a finding recorded where the next")
    print("person will meet it. That is a purpose, not an oversight. What is a defect is a")
    print("field written to satisfy a CHECK that the check does not read, which is what")
    print("happened to the empagliflozin assessment an hour ago.")


if __name__ == "__main__":
    main()
