def se_of(lo, hi):
    return (math.log(hi) - math.log(lo)) / (2 * Z95)


def pool(names, tau2_method="PM"):
    """Inverse-variance pool with Paule-Mandel tau-squared and a floored Knapp-Hartung interval.

    DL IS NOT USED AND THE REASON IS THE HOUSE RULE: DerSimonian-Laird is biased below k=10.
    The Knapp-Hartung interval carries the standard floor -- when Q is below its degrees of
    freedom the raw factor NARROWS the interval below the normal one, which is not what
    Knapp-Hartung is for.
    """
    th = [math.log(TRIALS[n]["hr"]) for n in names]
    se = [se_of(TRIALS[n]["lci"], TRIALS[n]["uci"]) for n in names]
    k = len(th)
    v = [s * s for s in se]
    tau2 = 0.0
    for _ in range(200):                       # Paule-Mandel, by fixed-point iteration
        w = [1.0 / (vi + tau2) for vi in v]
        mu = sum(wi * ti for wi, ti in zip(w, th)) / sum(w)
        q = sum(wi * (ti - mu) ** 2 for wi, ti in zip(w, th))
        if k <= 1:
            break
        new = tau2 + (q - (k - 1)) / sum(w)
        new = max(0.0, new)
        if abs(new - tau2) < 1e-12:
            tau2 = new
            break
        tau2 = new
    w = [1.0 / (vi + tau2) for vi in v]
    tot = sum(w)
    mu = sum(wi * ti for wi, ti in zip(w, th)) / tot
    # Q and I-squared are reported at tau-squared = 0, which is what they mean.
    w0 = [1.0 / vi for vi in v]
    mu0 = sum(wi * ti for wi, ti in zip(w0, th)) / sum(w0)
    Q = sum(wi * (ti - mu0) ** 2 for wi, ti in zip(w0, th))
    df = k - 1
    i2 = max(0.0, (Q - df) / Q) * 100 if Q > 0 else 0.0
    hk = max(1.0, sum(wi * (ti - mu) ** 2 for wi, ti in zip(w, th)) / df) / tot if df else None
    se_hk = math.sqrt(hk) if hk else math.sqrt(1.0 / tot)
    tq = {1: 12.706205, 2: 4.302653, 3: 3.182446, 4: 2.776445}.get(df, Z95)
    return {
        "k": k, "trials": list(names),
        "point": math.exp(mu),
        "ci_low": math.exp(mu - tq * se_hk), "ci_high": math.exp(mu + tq * se_hk),
        "tau2": tau2, "Q": Q, "df": df, "i2": i2,
        "estimator": "Paule-Mandel tau-squared, Knapp-Hartung t interval on %d df with the "
                     "variance-inflation floor at 1" % df,
        "n_total": sum(TRIALS[n]["n_total"] for n in names),
    }


def component_analysis():
    names = list(TRIALS)
    sets = {n: set(TRIALS[n]["components"]) for n in names}
    shared = set.intersection(*sets.values())
    allc = set.union(*sets.values())
    rows = []
    for c in sorted(allc):
        holders = sorted(n for n in names if c in sets[n])
        rows.append({"component": c, "canonical": CANONICAL.get(c, c),
                     "in_trials": holders, "in_all_three": len(holders) == len(names)})
    return {"shared_by_all_three": sorted(shared), "union": sorted(allc), "by_component": rows}

def selftest():
    fails = []

    def ck(name, got, want):
        ok = got == want
        print("  %-68s %s  %r" % (name, "ok" if ok else "FAIL", got))
        if not ok:
            fails.append(name)

    print("1. THE COMPONENT COMPARISON IS BY SET, NOT BY NAME (P37):")
    ca = component_analysis()
    ck("exactly ONE component is shared by all three", ca["shared_by_all_three"], ["cv_death"])
    ck("and the union is larger than any one trial's set", len(ca["union"]), 10)
    ck("resuscitated cardiac arrest is in COLCOT alone",
       [r["in_trials"] for r in ca["by_component"]
        if r["component"] == "resuscitated_cardiac_arrest"], [["COLCOT"]])

    print("\n2. THE THREE MI DEFINITIONS ARE THREE DIFFERENT COMPONENTS, not one:")
    mis = [r["component"] for r in ca["by_component"] if "myocardial_infarction" in r["component"]]
    ck("three distinct myocardial-infarction components", len(mis), 3)

    print("\n3. THE POOL REPRODUCES ITS OWN INPUTS -- a k=1 'pool' is the trial itself:")
    p1 = pool(["COLCOT"])
    ck("k=1 point is COLCOT's own hazard ratio", round(p1["point"], 4), 0.77)

    print("\n4. AND A k=2 POOL LIES INSIDE THE HULL OF ITS TWO TRIALS:")
    p2 = pool(["COLCOT", "CLEAR SYNERGY"])
    ck("0.77 <= pooled <= 0.99", 0.77 <= p2["point"] <= 0.99, True)

    print("\n5. EVERY EFFECT ESTIMATE IS ACCOMPANIED BY THE SENTENCE IT WAS READ FROM:")
    for n, t in TRIALS.items():
        ck("%s carries a verbatim effect sentence" % n,
           str(t["hr"]) in t["effect_verbatim"], True)
        ck("%s carries a PMID and a DOI" % n, bool(t["pmid"] and t["doi"]), True)

    print("\n6. THE ARITHMETIC OF EACH TRIAL'S OWN ARMS RECONCILES WITH ITS TOTAL:")
    for n, t in TRIALS.items():
        ck("%s: %d + %d == %d" % (n, t["n_treatment"], t["n_control"], t["n_total"]),
           t["n_treatment"] + t["n_control"], t["n_total"])

    print("\n%s" % ("SELFTEST FAILED: %s" % fails if fails else "SELFTEST PASSED"))
    return 1 if fails else 0
