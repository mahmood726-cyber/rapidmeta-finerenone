def main():
    sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8", errors="replace")
    from collections import Counter
    scr = json.load(io.open(os.path.join(REPO, "evidence", "2026-08-19-batch1",
                                          "apixaban_split_screening.json"), encoding="utf-8"))
    rc = 0
    out = {"adjudicated_utc": "2026-08-19", "boundary_rule": (
        "EXTENDED ANTICOAGULATION IN PATIENTS WHO HAVE ALREADY HAD A VTE IS TREATMENT; PRIMARY "
        "PROPHYLAXIS IN PATIENTS WHO HAVE NOT IS PREVENTION. The discriminator is whether the "
        "patient has already had the event, read from eligibility and description, because "
        "primaryPurpose does not carry it and both registrant conventions are used for both."),
        "topics": {}}

    for topic in (T, P):
        need = {r["nct"] for r in scr[topic]["trials"]
                if r["verdict"] == "NEEDS_ADJUDICATION"}
        mine = [(n, v, w) for (tp, n, v, w) in ROWS if tp == topic]
        got = {n for n, _v, _w in mine}
        missing, extra = sorted(need - got), sorted(got - need)
        print("--- %s: %d to adjudicate, %d rows" % (topic, len(need), len(mine)))
        if missing:
            print("   REFUSED: no row for %s" % missing)
            rc = 1
        if extra:
            print("   REFUSED: rows for trials not needing adjudication: %s" % extra)
            rc = 1
        tally = Counter(v for _n, v, _w in mine)
        for k, v in tally.most_common():
            print("   %-34s %2d" % (k, v))
        out["topics"][topic] = {"n": len(mine), "tally": dict(tally),
                                "trials": [{"nct": n, "verdict": v, "why": w}
                                           for n, v, w in mine]}
    if rc:
        return rc

    # The headline consequence, computed rather than asserted.
    for topic in (T, P):
        base = scr[topic]["tally"]["ELIGIBLE_POOLABLE_NOT_INCLUDED"]
        added = out["topics"][topic]["tally"].get("ELIGIBLE_POOLABLE_NOT_INCLUDED", 0)
        out["topics"][topic]["eligible_poolable_total"] = base + added
        print("\n%s: ELIGIBLE_POOLABLE_NOT_INCLUDED %d + %d adjudicated = %d"
              % (topic, base, added, base + added))
    with io.open(DEST, "w", encoding="utf-8") as fh:
        fh.write(json.dumps(out, indent=1))
    print("\nwrote %s" % DEST)
    print("\nREMAINDER AFTER ADJUDICATION: 0 on both reviews.")
    return 0


if __name__ == "__main__":
