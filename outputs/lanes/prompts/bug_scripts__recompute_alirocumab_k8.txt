def t_quantile(p, df):
    """Two-sided upper quantile of Student t, by bisection on the CDF.

    Written out rather than imported: scipy is not a declared dependency of this repo and a
    silent import failure here would produce a normal-quantile interval wearing a t label.
    """
    def betacf(a, b, x):
        MAXIT, EPS, FPMIN = 300, 3e-16, 1e-300
        qab, qap, qam = a + b, a + 1.0, a - 1.0
        c, d = 1.0, 1.0 - qab * x / qap
        if abs(d) < FPMIN:
            d = FPMIN
        d = 1.0 / d
        h = d
        for m in range(1, MAXIT + 1):
            m2 = 2 * m
            aa = m * (b - m) * x / ((qam + m2) * (a + m2))
            d = 1.0 + aa * d
            if abs(d) < FPMIN:
                d = FPMIN
            c = 1.0 + aa / c
            if abs(c) < FPMIN:
                c = FPMIN
            d = 1.0 / d
            h *= d * c
            aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2))
            d = 1.0 + aa * d
            if abs(d) < FPMIN:
                d = FPMIN
            c = 1.0 + aa / c
            if abs(c) < FPMIN:
                c = FPMIN
            d = 1.0 / d
            de = d * c
            h *= de
            if abs(de - 1.0) < EPS:
                break
        return h

    def betai(a, b, x):
        if x <= 0.0:
            return 0.0
        if x >= 1.0:
            return 1.0
        lbeta = (math.lgamma(a + b) - math.lgamma(a) - math.lgamma(b)
                 + a * math.log(x) + b * math.log(1.0 - x))
        if x < (a + 1.0) / (a + b + 2.0):
            return math.exp(lbeta) * betacf(a, b, x) / a
        return 1.0 - math.exp(lbeta) * betacf(b, a, 1.0 - x) / b

    def cdf(t):
        x = df / (df + t * t)
        tail = 0.5 * betai(df / 2.0, 0.5, x)
        return 1.0 - tail if t > 0 else tail

    lo, hi = 0.0, 200.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if cdf(mid) < p:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0


def dl(points, ses):
    k = len(points)
    w = [1.0 / (s * s) for s in ses]
    fe = sum(wi * yi for wi, yi in zip(w, points)) / sum(w)
    q = sum(wi * (yi - fe) ** 2 for wi, yi in zip(w, points))
    c = sum(w) - sum(wi * wi for wi in w) / sum(w)
    tau2 = max(0.0, (q - (k - 1)) / c)
    return pool(points, ses, tau2, q, k)


def reml(points, ses, iters=500):
    k = len(points)
    tau2 = 0.0
    for _ in range(iters):
        w = [1.0 / (s * s + tau2) for s in ses]
        mu = sum(wi * yi for wi, yi in zip(w, points)) / sum(w)
        # THE REML CORRECTION TERM IS ADDED AFTER THE DIVISION, NOT INSIDE THE NUMERATOR.
        # Written the wrong way round first, the iteration diverged until every weight
        # underflowed to zero and the next mean was 0/0 -- which is at least a LOUD failure.
        # tau2_{n+1} = sum(w^2[(y-mu)^2 - v]) / sum(w^2)  +  1/sum(w)   (Viechtbauer 2005)
        num = sum(wi ** 2 * ((yi - mu) ** 2 - s * s) for wi, yi, s in zip(w, points, ses))
        den = sum(wi ** 2 for wi in w)
        new = max(0.0, num / den + 1.0 / sum(w))
        if abs(new - tau2) < 1e-12:
            tau2 = new
            break
        tau2 = new
    w = [1.0 / (s * s) for s in ses]
    fe = sum(wi * yi for wi, yi in zip(w, points)) / sum(w)
    q = sum(wi * (yi - fe) ** 2 for wi, yi in zip(w, points))
    return pool(points, ses, tau2, q, k)


def pool(points, ses, tau2, q, k):
    w = [1.0 / (s * s + tau2) for s in ses]
    mu = sum(wi * yi for wi, yi in zip(w, points)) / sum(w)
    se_mu = math.sqrt(1.0 / sum(w))
    i2 = max(0.0, (q - (k - 1)) / q) * 100.0 if q > 0 else 0.0
    return {"k": k, "tau2": tau2, "q": q, "df": k - 1, "i2": i2,
            "point": mu, "se": se_mu,
            "ci_low": mu - Z * se_mu, "ci_high": mu + Z * se_mu}


def i2_metafor(ses, tau2):
    """metafor's I-squared: tau2 / (tau2 + s2), s2 the TYPICAL within-study variance.

    NOT the same quantity as Higgins (Q - df)/Q, and the difference is the whole reason this
    function exists. Higgins is computed from Q, which uses FIXED-EFFECT weights, so it does
    NOT depend on the estimator: it reads 88.0 for DL and 88.0 for REML. metafor's depends on
    tau2 and therefore does: 88.0 for DL and 90.0 for REML.

    A SENSITIVITY TABLE WHOSE I-SQUARED COLUMN CANNOT MOVE IS NOT A SENSITIVITY TABLE. The
    stored k=6 rows read 87.9 / 91.6 / 91.6, which only the metafor definition can produce,
    so that is the definition the column has always used and it is now named rather than
    assumed. The object's HEADLINE I-squared stays Higgins, where its own heterogeneity block
    declares it and gives the evidence for that choice.
    """
    k = len(ses)
    w = [1.0 / (s * s) for s in ses]
    sw, sw2 = sum(w), sum(wi * wi for wi in w)
    s2 = (k - 1) * sw / (sw * sw - sw2)
    return 100.0 * tau2 / (tau2 + s2) if (tau2 + s2) > 0 else 0.0


def hksj(points, ses, tau2, k):
    w = [1.0 / (s * s + tau2) for s in ses]
    mu = sum(wi * yi for wi, yi in zip(w, points)) / sum(w)
    # HKSJ VARIANCE INFLATION, WITH THE FLOOR. Where Q < k-1 the raw factor is BELOW 1 and
    # would NARROW the interval below DerSimonian-Laird, which is the opposite of what the
    # method is for. Floored at 1 per this project's standing rule.
    raw = sum(wi * (yi - mu) ** 2 for wi, yi in zip(w, points)) / ((k - 1) * sum(w))
    var = max(raw, 1.0 / sum(w))
    se = math.sqrt(var)
    t = t_quantile(0.975, k - 1)
    return {"point": mu, "se": se, "t": t, "df": k - 1,
            "ci_low": mu - t * se, "ci_high": mu + t * se}
