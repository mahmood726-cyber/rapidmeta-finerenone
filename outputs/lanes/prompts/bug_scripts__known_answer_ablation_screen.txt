    A SCREENER TESTED ONLY ON TRIALS IT WAS WRITTEN AGAINST IS TESTED AGAINST ITS AUTHOR.
    The five sibling instances were decided by hand, on a different day's reasoning, for a
    review with a NARROWER population -- so they are an outside answer, not a restatement of

The population limb DIFFERS between the two reviews (this one does not require heart failure),

def main():
    sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8", errors="replace")
    failures = []
    for nct, what, expected in CASES:
        st, study, det = X.fetch_raw(nct)
        if st != X.OK:
            print("%s  UNREACHABLE (%s) -- NOT_ASSESSABLE, and a known-answer test that "
                  "cannot read its instance has not passed" % (nct, det))
            failures.append((nct, "unreachable"))
            continue
        doc = X.require_raw_v2(study, nct)
        _v, _axis, _crit, limbs, kinds, _status, _conds, _reason = S.screen(nct, doc)
        bad = {k: (limbs.get(k), v) for k, v in expected.items() if limbs.get(k) != v}
        print("%s  %s" % (nct, "OK  " if not bad else "WRONG"))
        print("      %s" % what)
        print("      arms %s -> %s" % (kinds, limbs))
        if bad:
            for k, (got, want) in bad.items():
                print("      %s: expected %s, got %s" % (k, want, got))
            failures.append((nct, bad))
        print()

    print("instances %d, failures %d" % (len(CASES), len(failures)))
    if failures:
        print("\nREFUSED: the screener disagrees with answers established independently of it. "
              "The\nrules are wrong, or the term sets are, and a 621-trial pass must not run "
              "on them.")
        return 1
    print("\nAll instances agree. The screener may be run over the remainder.")
    print("WHAT THIS DOES NOT ESTABLISH: eight instances cannot show the term sets are "
          "complete.\nThat is what NEEDS_ADJUDICATION is for -- it is the screener saying so "
          "rather than guessing.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
