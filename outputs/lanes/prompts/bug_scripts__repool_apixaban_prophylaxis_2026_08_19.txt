def rr_pool(rows):
    """DerSimonian-Laird on log risk ratio. Events derived from rate x n, rounded."""
    ys, ses, per = [], [], []
    for r in rows:
        acr, ap, ep, na, ne = r[0], r[-5], r[-4], r[-3], r[-2]
        ea, ee = round(ap * na / 100.0), round(ep * ne / 100.0)
        # 0.5 continuity correction ONLY if a cell is zero -- unconditional correction biases
        # the ratio toward 1 (rules/advanced-stats).
        a, b = (ea, ee) if (ea and ee) else (ea + 0.5, ee + 0.5)
        na2, ne2 = (na, ne) if (ea and ee) else (na + 1, ne + 1)
        y = math.log((a / na2) / (b / ne2))
        se = math.sqrt(1.0 / a - 1.0 / na2 + 1.0 / b - 1.0 / ne2)
        ys.append(y)
        ses.append(se)
        per.append({"trial": acr, "events_apixaban": ea, "n_apixaban": na,
                    "events_enoxaparin": ee, "n_enoxaparin": ne,
                    "rr": round(math.exp(y), 4),
                    "ci_low": round(math.exp(y - Z * se), 4),
                    "ci_high": round(math.exp(y + Z * se), 4)})
    k = len(ys)
    w = [1.0 / s ** 2 for s in ses]
    fe = sum(a * b for a, b in zip(w, ys)) / sum(w)
    q = sum(a * (b - fe) ** 2 for a, b in zip(w, ys))
    c = sum(w) - sum(x * x for x in w) / sum(w)
    tau2 = max(0.0, (q - (k - 1)) / c)
    w2 = [1.0 / (s ** 2 + tau2) for s in ses]
    mu = sum(a * b for a, b in zip(w2, ys)) / sum(w2)
    se = math.sqrt(1.0 / sum(w2))
    i2 = max(0.0, (q - (k - 1)) / q) * 100.0 if q > 0 else 0.0
    return {"k": k, "rr": round(math.exp(mu), 4),
            "ci_low": round(math.exp(mu - Z * se), 4),
            "ci_high": round(math.exp(mu + Z * se), 4),
            "q": round(q, 4), "df": k - 1, "tau2": round(tau2, 5),
            "i2_pct": round(i2, 1), "per_trial": per}
